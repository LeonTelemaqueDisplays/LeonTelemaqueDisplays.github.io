Technical Artifact
SQL — Cohort Churn Risk Scoring Query Portfolio Artifact · Production-Grade
-- GTA Online: Pre-GTA VI Churn Risk Scoring · Operation Last Ride -- Segments 18.3M MAU by behavioral signals, 90-day lookback WITH session_metrics AS ( SELECT player_id, platform, -- 'console' | 'pc' COUNT(*) AS sessions_90d, AVG(session_minutes) AS avg_session_len, SUM(real_money_spend_usd) AS total_spend_90d, DATEDIFF('2026-06-04', MAX(login_date)) AS days_since_login FROM player_sessions WHERE session_date BETWEEN '2026-03-06' AND '2026-06-04' GROUP BY 1, 2 ), churn_scored AS ( SELECT player_id, platform, ROUND( /* Platform: console players migrate to GTA VI on launch day */ (CASE WHEN platform = 'console' THEN 1.0 ELSE 0.2 END) * 0.30 + /* Session frequency decay — see session_decay.html for full model */ (CASE WHEN sessions_90d < 6 THEN 1.0 WHEN sessions_90d < 12 THEN 0.6 ELSE 0.2 END) * 0.25 + /* Recency */ (CASE WHEN days_since_login > 30 THEN 1.0 WHEN days_since_login > 14 THEN 0.5 ELSE 0.1 END) * 0.25 + /* Monetization depth — see economy_monetization.html for Shark Card model */ (CASE WHEN total_spend_90d > 50 THEN -0.20 ELSE 0 END) , 3) AS churn_score FROM session_metrics ) SELECT CASE WHEN churn_score >= 0.75 THEN 'CRITICAL' WHEN churn_score >= 0.45 THEN 'HIGH' ELSE 'MONITOR' END AS risk_tier, COUNT(*) AS player_count, ROUND(AVG(churn_score), 3) AS avg_score, CASE risk_tier WHEN 'CRITICAL' THEN 'Last Ride Event + GTA+ Bridge Bundle' WHEN 'HIGH' THEN 'FOMO Event + Win-Back Push' ELSE 'Monitor + Standard Content Calendar' END AS recommended_lever FROM churn_scored GROUP BY 1 ORDER BY avg_score DESC;
