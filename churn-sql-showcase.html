Frustration Score — Per-Session Computation
Composite signal: loss streak + rage quit flag + match intensity + session exit type
Window Fn CASE Aggregate 
-- Frustration score per session (0-100 composite signal)
-- Used by Streamlit dashboard to colour-code player health tiles

WITH session_signals AS (
  SELECT
    ps.session_id,
    ps.player_id,
    ps.session_start,
    ps.matches_played,
    ps.rage_quit,
    ps.exit_type,
    ps.match_intensity_score,

    -- Running max loss streak within session
    MAX(me.loss_streak_count) AS peak_loss_streak,

    -- Count of rage quit events in session
    COUNT(me.event_id) FILTER (
      WHERE me.event_type = 'rage_quit'
    ) AS rq_count,

    -- Win rate for the session
    ROUND(
      ps.matches_won::NUMERIC / NULLIF(ps.matches_played, 0), 2
    ) AS session_win_rate

  FROM player_sessions ps
  JOIN match_events me
    ON me.session_id = ps.session_id
  GROUP BY
    ps.session_id, ps.player_id, ps.session_start,
    ps.matches_played, ps.matches_won,
    ps.rage_quit, ps.exit_type, ps.match_intensity_score
),

scored AS (
  SELECT
    session_id, player_id, session_start,
    peak_loss_streak, rq_count, session_win_rate,

    -- Composite frustration score (0–100)
    LEAST(100, (
        -- Loss streak contribution (0–40 pts)
        LEAST(40, peak_loss_streak * 8)

        -- Rage quit presence (25 pts)
        + CASE WHEN rq_count > 0 THEN 25 ELSE 0 END

        -- Multiple rage quits (extra 10 pts)
        + CASE WHEN rq_count > 1 THEN 10 ELSE 0 END

        -- Low win rate when played >=3 matches (0–20 pts)
        + CASE
            WHEN matches_played >= 3 AND session_win_rate < 0.25 THEN 20
            WHEN matches_played >= 3 AND session_win_rate < 0.40 THEN 10
            ELSE 0
          END

        -- High match intensity pressure (0–5 pts)
        + CASE WHEN match_intensity_score > 8.0 THEN 5 ELSE 0 END
    )) AS frustration_score

  FROM session_signals
)

SELECT
  session_id,
  player_id,
  session_start::DATE AS session_date,
  peak_loss_streak,
  rq_count,
  ROUND(session_win_rate * 100, 1) AS win_rate_pct,
  frustration_score,
  CASE
    WHEN frustration_score >= 70 THEN 'CRITICAL'
    WHEN frustration_score >= 45 THEN 'HIGH'
    WHEN frustration_score >= 20 THEN 'MODERATE'
    ELSE 'LOW'
  END AS frustration_band
FROM scored
ORDER BY frustration_score DESC;

Design note: The frustration score is intentionally a weighted composite rather than a single signal. A player with a 7-game loss streak but no rage quits is a different intervention case than one who rage-quit twice in 3 matches. The banding (CRITICAL / HIGH / MODERATE / LOW) maps directly to LiveOps intervention tiers — CRITICAL players are flagged for matchmaking adjustment the following session.
02 · D1 / D7 / D30 Cohort Retention
Cohort Retention Matrix — Monthly Cohorts
Tracks what % of each monthly cohort returned on D1, D7, D14, D30, D60, D90
DATE_TRUNCFILTERCohort
-- Monthly cohort retention matrix
-- Powers the Tableau heatmap and Streamlit retention curve

WITH cohort_base AS (
  SELECT
    p.player_id,
    DATE_TRUNC('month', p.first_play_ts) AS cohort_month,
    p.first_play_ts AS install_ts
  FROM players p
),

player_days AS (
  SELECT
    cb.player_id,
    cb.cohort_month,
    DATE_TRUNC('day', ps.session_start) AS active_day,
    (DATE_TRUNC('day', ps.session_start)
      - DATE_TRUNC('day', cb.install_ts))::INT AS day_number
  FROM cohort_base cb
  JOIN player_sessions ps
    ON ps.player_id = cb.player_id
)

SELECT
  cb.cohort_month,
  COUNT(DISTINCT cb.player_id)                                    AS cohort_size,

  -- D1 retention
  COUNT(DISTINCT pd.player_id) FILTER (WHERE pd.day_number BETWEEN 1 AND 1)
    * 100.0 / COUNT(DISTINCT cb.player_id)                         AS d1_retention_pct,

  -- D7 retention
  COUNT(DISTINCT pd.player_id) FILTER (WHERE pd.day_number BETWEEN 6 AND 8)
    * 100.0 / COUNT(DISTINCT cb.player_id)                         AS d7_retention_pct,

  -- D30 retention
  COUNT(DISTINCT pd.player_id) FILTER (WHERE pd.day_number BETWEEN 28 AND 32)
    * 100.0 / COUNT(DISTINCT cb.player_id)                         AS d30_retention_pct

FROM cohort_base cb
LEFT JOIN player_days pd
  ON pd.player_id = cb.player_id
GROUP BY cb.cohort_month
ORDER BY cb.cohort_month;

03 · Loss Streak Detection — Window Function
Running Loss Streak Counter per Player
Uses LAG() to detect consecutive losses and reset the counter on a win
LAG()SUM() OVERPARTITION BY
-- Running loss streak per player, ordered by event timestamp
-- Resets to 0 on any win — used to trigger real-time intervention flags

WITH match_results AS (
  SELECT
    player_id,
    event_ts,
    match_result,
    CASE WHEN match_result = 'loss' THEN 1 ELSE 0 END AS is_loss,

    -- Group counter: increments every time a win breaks the streak
    SUM(CASE WHEN match_result != 'loss' THEN 1 ELSE 0 END)
      OVER (
        PARTITION BY player_id
        ORDER BY event_ts
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
      ) AS win_group

  FROM match_events
  WHERE event_type IN ('match_end', 'rage_quit')
),

streaks AS (
  SELECT
    player_id,
    event_ts,
    match_result,
    is_loss,
    win_group,

    -- Streak = running sum of losses within each win_group
    SUM(is_loss) OVER (
      PARTITION BY player_id, win_group
      ORDER BY event_ts
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS current_streak

  FROM match_results
)

SELECT
  player_id,
  event_ts,
  match_result,
  current_streak,
  CASE
    WHEN current_streak >= 5 THEN '🔴 CRITICAL — intervene next session'
    WHEN current_streak >= 3 THEN '🟡 FLAGGED — monitor closely'
    ELSE                          '🟢 NORMAL'
  END AS intervention_flag
FROM streaks
ORDER BY player_id, event_ts;

Why window functions here: A simple GROUP BY player_id would only give us the total losses — it can't tell us that losses happened consecutively. The win_group trick (incrementing a counter on every non-loss) creates natural streak segments within each player's timeline, letting the second window function sum losses within each segment. This is a standard pattern for session-level streak analysis in LiveOps telemetry.
